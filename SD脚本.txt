-- ==================== 去除恐怖音频（增强版） ====================
local antiHorrorAudioEnabled = false
local horrorAudioConnections = {}
local mutedSounds = {}
local soundListeners = {}
local horrorKeywords = {
    "scary", "horror", "scream", "jump", "fear", "terrify", "creepy", "spooky",
    "demon", "ghost", "monster", "zombie", "blood", "death", "kill", "evil",
    "nightmare", "whisper", "distort", "haunt", "possess",
    "恐怖", "惊悚", "尖叫", "鬼", "恶魔", "幽灵", "怪物", "僵尸", "血腥",
    "死亡", "杀死", "邪恶", "噩梦", "低语", "失真", "吓"
}

local function isHorrorSound(sound)
    if not sound or not sound:IsA("Sound") then return false end
    local n = sound.Name
    if n == "SD_Music_1" or n == "SD_Music_3" or n == "SD_Music_4" or n == "SD_Music_5" then
        return false
    end
    if string.sub(n, 1, 13) == "SD_Intercept_" then return false end
    local lowerName = string.lower(n)
    for _, kw in ipairs(horrorKeywords) do
        if string.find(lowerName, kw) then return true end
    end
    if sound.Volume > 1 then return true end
    return false
end

local function disableHorrorSound(sound)
    if not antiHorrorAudioEnabled then return end
    if not sound or not sound:IsA("Sound") then return end
    if not isHorrorSound(sound) then return end
    if not mutedSounds[sound] then
        mutedSounds[sound] = { Volume = sound.Volume, Playing = sound.Playing }
    end
    sound.Volume = 0
    sound.Playing = false
end

local function attachSoundListener(sound)
    if not sound or not sound:IsA("Sound") then return end
    if soundListeners[sound] then return end
    local conns = {}
    table.insert(conns, sound:GetPropertyChangedSignal("Playing"):Connect(function()
        if antiHorrorAudioEnabled and sound.Playing then
            disableHorrorSound(sound)
        end
    end))
    table.insert(conns, sound:GetPropertyChangedSignal("Volume"):Connect(function()
        if antiHorrorAudioEnabled then
            disableHorrorSound(sound)
        end
    end))
    soundListeners[sound] = conns
    if sound.Playing then
        disableHorrorSound(sound)
    end
end

local function detachSoundListener(sound)
    if soundListeners[sound] then
        for _, c in ipairs(soundListeners[sound]) do
            pcall(function() c:Disconnect() end)
        end
        soundListeners[sound] = nil
    end
end

local function restoreAllMutedSounds()
    for sound, data in pairs(mutedSounds) do
        if sound and sound.Parent then
            pcall(function()
                sound.Volume = data.Volume
                if data.Playing then sound:Play() end
            end)
        end
    end
    mutedSounds = {}
end

local function detachAllListeners()
    for sound, _ in pairs(soundListeners) do
        detachSoundListener(sound)
    end
    soundListeners = {}
end

local function toggleAntiHorrorAudio(state)
    antiHorrorAudioEnabled = state
    if state then
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("Sound") then attachSoundListener(obj) disableHorrorSound(obj) end
        end
        for _, obj in ipairs(SoundService:GetDescendants()) do
            if obj:IsA("Sound") then attachSoundListener(obj) disableHorrorSound(obj) end
        end
        for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
            if obj:IsA("Sound") then attachSoundListener(obj) disableHorrorSound(obj) end
        end

        local connections = {}
        table.insert(connections, Workspace.DescendantAdded:Connect(function(desc)
            if antiHorrorAudioEnabled and desc:IsA("Sound") then
                attachSoundListener(desc)
                disableHorrorSound(desc)
            end
        end))
        table.insert(connections, SoundService.DescendantAdded:Connect(function(desc)
            if antiHorrorAudioEnabled and desc:IsA("Sound") then
                attachSoundListener(desc)
                disableHorrorSound(desc)
            end
        end))
        table.insert(connections, ReplicatedStorage.DescendantAdded:Connect(function(desc)
            if antiHorrorAudioEnabled and desc:IsA("Sound") then
                attachSoundListener(desc)
                disableHorrorSound(desc)
            end
        end))
        table.insert(connections, Workspace.DescendantRemoving:Connect(function(desc)
            if desc:IsA("Sound") then
                detachSoundListener(desc)
                mutedSounds[desc] = nil
            end
        end))
        horrorAudioConnections = connections

        task.spawn(function()
            while antiHorrorAudioEnabled do
                task.wait(1)
                for _, obj in ipairs(Workspace:GetDescendants()) do
                    if obj:IsA("Sound") then
                        if not soundListeners[obj] then attachSoundListener(obj) end
                        if obj.Playing then disableHorrorSound(obj) end
                    end
                end
                for _, obj in ipairs(SoundService:GetDescendants()) do
                    if obj:IsA("Sound") then
                        if not soundListeners[obj] then attachSoundListener(obj) end
                        if obj.Playing then disableHorrorSound(obj) end
                    end
                end
            end
        end)

    else
        for _, conn in ipairs(horrorAudioConnections) do
            pcall(function() conn:Disconnect() end)
        end
        horrorAudioConnections = {}
        detachAllListeners()
        restoreAllMutedSounds()
    end
end

local function removeESP(target)
    if not target then return end
    if espObjects[target] then
        if target:FindFirstChild("ESP_Highlight") then target.ESP_Highlight:Destroy() end
        local head = target:FindFirstChild("Head")
        if head and head:FindFirstChild("ESP_Billboard") then head.ESP_Billboard:Destroy() end
        espObjects[target] = nil
    end
end

local function applyESP(character)
    if not character or not character:IsA("Model") or character == LocalPlayer.Character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")
    if not humanoid or not head then return end
    local isPlayer = Players:GetPlayerFromCharacter(character) ~= nil
    local needHighlight = (isPlayer and playerEspEnabled) or (not isPlayer and npcEspEnabled)
    local needName = (isPlayer and playerNameEnabled) or (not isPlayer and npcNameEnabled)
    if not needHighlight and not needName then removeESP(character) return end
    removeESP(character)
    if needHighlight then
        local highlight = Instance.new("Highlight")
        highlight.Name = "ESP_Highlight"
        highlight.Adornee = character
        highlight.FillColor = isPlayer and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(0, 0, 255)
        highlight.FillTransparency = 0.5
        highlight.OutlineColor = isPlayer and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(0, 0, 255)
        highlight.OutlineTransparency = 0
        highlight.Parent = character
    end
    if needName then
        local billboard = Instance.new("BillboardGui")
        billboard.Name = "ESP_Billboard"
        billboard.Adornee = head
        billboard.Size = UDim2.new(0, 100, 0, 30)
        billboard.StudsOffset = Vector3.new(0, 2.5, 0)
        billboard.AlwaysOnTop = true
        billboard.Parent = head
        local textLabel = Instance.new("TextLabel")
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.TextColor3 = isPlayer and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(0, 0, 255)

        textLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)

        textLabel.TextSize = 14
        if isPlayer then
            local player = Players:GetPlayerFromCharacter(character)
            textLabel.Text = player.DisplayName .. " (@" .. player.Name .. ")"
        else
            textLabel.Text = "NPC"
        end
        textLabel.Parent = billboard
    end
    espObjects[character] = true
end

-- ==================== 密码扫描逻辑 ====================
local function scanForPasswords()
    local foundPasswords = {}
    local keywords = {"password", "pass", "code", "key", "门", "密码", "锁", "解锁", "door"}
    local function checkKeyword(text)
        if not text then return false end
        text = string.lower(tostring(text))
        for _, kw in ipairs(keywords) do
            if string.find(text, kw) then return true end
        end
        return false
    end
    local function scanObject(obj)
        if not obj then return end
        if obj:IsA("StringValue") or obj:IsA("NumberValue") then
            if checkKeyword(obj.Name) or checkKeyword(obj.Value) then
                table.insert(foundPasswords, "[" .. obj.ClassName .. "] " .. obj:GetFullName() .. " = " .. tostring(obj.Value))
            end
        end
        if obj:IsA("TextLabel") or obj:IsA("TextBox") or obj:IsA("TextButton") then
            if checkKeyword(obj.Name) or checkKeyword(obj.Text) then
                table.insert(foundPasswords, "[" .. obj.ClassName .. "] " .. obj:GetFullName() .. " 内容: " .. tostring(obj.Text))
            end
        end
    end
    for _, obj in ipairs(Workspace:GetDescendants()) do scanObject(obj) end
    for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do scanObject(obj) end
    return foundPasswords
end

-- ==================== 导航红线绘制（透视+高光增强版） ====================
local function drawLine(p1, p2)
    if not navFolder or not navFolder.Parent then
        navFolder = Instance.new("Folder")
        navFolder.Name = "SD_NavLines"
        navFolder.Parent = Workspace
    end
    local dist = (p2 - p1).Magnitude
    if dist < 0.1 then return end

    local part = Instance.new("Part")
    part.Anchored = true
    part.CanCollide = false
    part.CanQuery = false
    part.CanTouch = false
    part.Material = Enum.Material.Neon
    part.Color = Color3.fromRGB(255, 30, 30)
    part.Transparency = 0
    part.Size = Vector3.new(0.35, 0.35, dist)
    part.CFrame = CFrame.new(p1, p2) * CFrame.new(0, 0, -dist/2)
    part.Parent = navFolder

    local highlight = Instance.new("Highlight")
    highlight.Name = "NavHighlight"
    highlight.Adornee = part
    highlight.FillColor = Color3.fromRGB(255, 0, 0)
    highlight.FillTransparency = 0.7
    highlight.OutlineColor = Color3.fromRGB(255, 100, 100)
    highlight.OutlineTransparency = 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Parent = part

    local light = Instance.new("PointLight")
    light.Color = Color3.fromRGB(255, 50, 50)
    light.Brightness = 2
    light.Range = 8
    light.Parent = part
end

local function startNavigation(targetPart)
    clearNavLines()
    isNavigating = true
    if navTask then task.cancel(navTask) end
    navTask = task.spawn(function()
        while isNavigating do
            task.wait(0.5)
            if not isNavigating then break end
            local char = LocalPlayer.Character
            if not char then break end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if not hrp or not humanoid then break end
            if not targetPart or not targetPart.Parent then clearNavLines() break end
            local distToTarget = (hrp.Position - targetPart.Position).Magnitude
            if distToTarget < 10 then clearNavLines() break end
            local path = PathfindingService:CreatePath({AgentRadius = 2, AgentHeight = 5, AgentCanJump = true})
            local success = pcall(function() path:ComputeAsync(hrp.Position, targetPart.Position) end)
            if success and path.Status == Enum.PathStatus.Success then
                if navFolder and navFolder.Parent then
                    for _, part in ipairs(navFolder:GetChildren()) do part:Destroy() end
                end
                local waypoints = path:GetWaypoints()
                for i = 1, #waypoints - 1 do
                    drawLine(waypoints[i].Position + Vector3.new(0, 2, 0), waypoints[i+1].Position + Vector3.new(0, 2, 0))
                end
            else
                if navFolder and navFolder.Parent then
                    for _, part in ipairs(navFolder:GetChildren()) do part:Destroy() end
                end
                drawLine(hrp.Position + Vector3.new(0, 2, 0), targetPart.Position + Vector3.new(0, 2, 0))
            end
        end
    end)
end

local function scanForInteractiveTeleports()
    local results = {}

    local allowKeywords = {
        "achievement", "achievements", "achievementpoint", "成就", "成就点",
        "teleport", "teleportation", "传送", "传送点", "传送门", "portal", "warp",
        "checkpoint", "检查点", "存档点", "savepoint",
        "exit", "出口", "next", "下一关", "下一层", "关卡", "过关", "通关",
        "level", "stage", "reward", "奖励"
    }

    local denyKeywords = {
        "door", "门", "gate", "闸", "button", "按钮", "开关",
        "trap", "陷阱", "lever", "拉杆", "switch", "brick"
    }

    local function hasAllowKeyword(name)
        name = string.lower(name)
        for _, kw in ipairs(allowKeywords) do
            if string.find(name, kw) then return true end
        end
        return false
    end

    local function hasDenyKeyword(name)
        name = string.lower(name)
        for _, kw in ipairs(denyKeywords) do
            if string.find(name, kw) then return true end
        end
        return false
    end

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local name = obj.Name
            if hasAllowKeyword(name) and not hasDenyKeyword(name) then
                table.insert(results, obj)
            end
        end
    end
    return results
end

local nightVisionEnabled = false
local nightVisionConnection = nil
local defaultAmbient = Lighting.Ambient
local defaultOutdoor = Lighting.OutdoorAmbient
local ORARIO_NOTTE_INIZIO = 18
local ORARIO_NOTTE_FINE = 6
local nightVisionColor = Color3.fromRGB(255, 255, 255)

local function startNightVision()
    defaultAmbient = Lighting.Ambient
    defaultOutdoor = Lighting.OutdoorAmbient
    if nightVisionConnection then nightVisionConnection:Disconnect() end
    nightVisionConnection = RunService.Heartbeat:Connect(function()
        if not nightVisionEnabled then return end
        local tempo = Lighting.ClockTime
        if tempo >= ORARIO_NOTTE_INIZIO or tempo < ORARIO_NOTTE_FINE then
            Lighting.Ambient = nightVisionColor
            Lighting.OutdoorAmbient = nightVisionColor
        else
            Lighting.Ambient = defaultAmbient
            Lighting.OutdoorAmbient = defaultOutdoor
        end
    end)
end

local function stopNightVision()
    if nightVisionConnection then nightVisionConnection:Disconnect() nightVisionConnection = nil end
    Lighting.Ambient = defaultAmbient
    Lighting.OutdoorAmbient = defaultOutdoor
end

                local dist = (hrp.Position - primary.Position).Magnitude
                if dist < 20 and model ~= char and not model:IsDescendantOf(char) then
                    model:Destroy()
                    showJumpScareWarning()
                    playRandomInterceptMusic()
                    loadRandomInterceptScript()
                end
            end
        elseif model:IsA("BasePart") then
            local dist = (hrp.Position - model.Position).Magnitude
            if dist < 20 and model ~= char and not model:IsDescendantOf(char) then
                model:Destroy()
                showJumpScareWarning()
                playRandomInterceptMusic()
                loadRandomInterceptScript()
            end
        end
    end
end

local freecamEnabled = false

local freecamSpeed = 50
local freecamRotX, freecamRotY = 0, 0
local freecamCamera = Workspace.CurrentCamera
local originalCameraType = freecamCamera.CameraType
local originalCameraCFrame = freecamCamera.CFrame
local freecamSavedJumpPower = nil
local freecamSavedJumpHeight = nil

local function disableCharacterMovement()
    local char = LocalPlayer.Character
    if not char then return end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    if freecamSavedJumpPower == nil then
        freecamSavedJumpPower = humanoid.JumpPower
        freecamSavedJumpHeight = humanoid.JumpHeight
    end
    humanoid.WalkSpeed = 0
    humanoid.JumpPower = 0
    humanoid.JumpHeight = 0
end

local function restoreCharacterMovement()
    local char = LocalPlayer.Character
    if not char then return end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    humanoid.WalkSpeed = currentWalkSpeed
    if freecamSavedJumpPower then humanoid.JumpPower = freecamSavedJumpPower end
    if freecamSavedJumpHeight then humanoid.JumpHeight = freecamSavedJumpHeight end
    freecamSavedJumpPower = nil
    freecamSavedJumpHeight = nil
end

local function toggleFreecam(state)
    freecamEnabled = state
    if state then
        -- 禁用角色移动（避免人物被 WASD 控制）
        disableCharacterMovement()

        originalCameraType = freecamCamera.CameraType
        originalCameraCFrame = freecamCamera.CFrame
        freecamCamera.CameraType = Enum.CameraType.Scriptable
        local look = freecamCamera.CFrame.LookVector
        freecamRotY = math.atan2(-look.X, -look.Z)
        freecamRotX = math.asin(look.Y)

        freecamConnection = RunService.RenderStepped:Connect(function(dt)
            if not freecamEnabled then return end
            if UserInputService:GetFocusedTextBox() then return end
            local cam = Workspace.CurrentCamera
            if not cam then return end

            -- 持续禁用角色移动（防止游戏内部重置）
            local char = LocalPlayer.Character
            if char then
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.WalkSpeed ~= 0 then
                    humanoid.WalkSpeed = 0
                end
            end

            local moveDir = Vector3.new(0, 0, 0)
            local camCFrame = cam.CFrame
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + camCFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - camCFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - camCFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + camCFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end

            if moveDir.Magnitude > 0 then
                cam.CFrame = cam.CFrame + moveDir.Unit * freecamSpeed * dt
            end

            if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
                UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
                UserInputService.MouseIconEnabled = false
                local delta = UserInputService:GetMouseDelta()
                freecamRotY = freecamRotY - delta.X * 0.005
                freecamRotX = math.clamp(freecamRotX - delta.Y * 0.005, -math.pi/2 + 0.01, math.pi/2 - 0.01)
                local newCFrame = CFrame.new(cam.CFrame.Position) * CFrame.Angles(0, freecamRotY, 0) * CFrame.Angles(freecamRotX, 0, 0)
                cam.CFrame = newCFrame
            else
                UserInputService.MouseBehavior = Enum.MouseBehavior.Default
                UserInputService.MouseIconEnabled = true
            end
        end)
    else
        if freecamConnection then
            freecamConnection:Disconnect()
            freecamConnection = nil
        end
        freecamCamera.CameraType = originalCameraType
        freecamCamera.CFrame = originalCameraCFrame
        UserInputService.MouseBehavior = Enum.MouseBehavior.Default
        UserInputService.MouseIconEnabled = true

        -- 恢复角色移动
        restoreCharacterMovement()
    end
end

    if freecamEnabled then
        disableCharacterMovement()
    end

local function translateText(text)
    if not text or text == "" then return nil end
    if translations[text] then return translations[text] end
    for eng, chn in pairs(translations) do
        if #eng > 4 and string.find(text, eng, 1, true) then
            return text:gsub(eng, chn)
        end
    end
    return nil
end

local function translateGUI(parent)
    if not parent then return end
    for _, obj in ipairs(parent:GetDescendants()) do
        pcall(function()
            if obj:IsA("TextLabel") or obj:IsA("TextButton") then
                local newText = translateText(obj.Text)
                if newText then obj.Text = newText end
            end
            if obj:IsA("TextBox") then
                local newPlaceholder = translateText(obj.PlaceholderText)
                if newPlaceholder then obj.PlaceholderText = newPlaceholder end
                local newText = translateText(obj.Text)
                if newText then obj.Text = newText end
            end
        end)
    end
end

local function refreshCodeDisplay()
    for _, child in ipairs(PwdOutput:GetChildren()) do
        if child:IsA("TextLabel") then child:Destroy() end
    end
    local results = scanForPasswords()
    if #results == 0 then
        local emptyLabel = Instance.new("TextLabel")
        emptyLabel.Size = UDim2.new(1, 0, 0, 30)
        emptyLabel.BackgroundTransparency = 1
        emptyLabel.Text = "未在当前地图找到包含密码关键词的内容。"
        emptyLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
        emptyLabel.TextSize = 12
        emptyLabel.Font = Enum.Font.Gotham
        emptyLabel.TextXAlignment = Enum.TextXAlignment.Left
        emptyLabel.Parent = PwdOutput
        PwdStatus.Text = "未找到匹配项"
        PwdStatus.TextColor3 = Color3.fromRGB(200, 100, 100)
    else
        for i, text in ipairs(results) do
            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, 0, 0, 25)
            label.BackgroundTransparency = 1
            label.Text = "• " .. text
            label.TextColor3 = Color3.fromRGB(0, 255, 150)
            label.TextSize = 12
            label.Font = Enum.Font.Code
            label.TextXAlignment = Enum.TextXAlignment.Left
            label.TextWrapped = true
            label.LayoutOrder = i
            label.Parent = PwdOutput
        end
        PwdStatus.Text = "找到 " .. #results .. " 个匹配项"
        PwdStatus.TextColor3 = Color3.fromRGB(0, 255, 150)
    end
    PwdOutput.CanvasSize = UDim2.new(0, 0, 0, #results * 30 + 10)
end

local originalDestroy = Fluent.Destroy
Fluent.Destroy = function(self)
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj.Name == "SD_NavLines" then obj:Destroy() end
    end
    clearAllAntiKickHighlights()
    if nightVisionConnection then nightVisionConnection:Disconnect() end
    stopAllMusic()
    for _, s in ipairs(interceptSounds) do s:Stop() end
    if nextbotConnection then nextbotConnection:Disconnect() end
    if noclipConnection then noclipConnection:Disconnect() end
    if freecamConnection then freecamConnection:Disconnect() end
    detachAllListeners()
    if warnBlinkConnection then warnBlinkConnection:Disconnect() end
    if jumpScareWarnGui then jumpScareWarnGui:Destroy() end
    if MiniGui then MiniGui:Destroy() end
    originalDestroy(self)
end
